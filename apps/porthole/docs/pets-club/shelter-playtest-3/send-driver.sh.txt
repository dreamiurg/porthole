#!/bin/bash
# usage: send.sh "cmd1" "cmd2" ... ; also expands tap X Y [ms] and taps
S=/tmp/claude-0/-home-user-porthole/4e7b951a-0e4f-5eae-859d-2ab6ed697cd9/scratchpad/sock
out=()
for c in "$@"; do
  set -- $c
  case $1 in
    tap) ms=${4:-80}; out+=("down $2 $3"); [ $ms -gt 33 ] && out+=("tick $((ms-33))"); out+=("up");;
    *) out+=("$c");;
  esac
done
printf '%s\n' "${out[@]}" | python3 -c "
import socket,sys
s=socket.socket(socket.AF_UNIX);s.connect('$S');f=s.makefile('rw')
for l in sys.stdin:
    f.write(l);f.flush();r=f.readline().strip()
    if r!='ok': print(l.strip(),'->',r)
"
