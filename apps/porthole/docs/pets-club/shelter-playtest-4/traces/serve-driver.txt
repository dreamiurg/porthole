"""Replay a visible-input trace against snap --serve from the fixture. Usage: drive.py TRACE OUTDIR [--keep]
Trace lines: down X Y | move X Y | up | tick MS | frame NAME | press X Y MS [EVERY] (down, ticks with frames every EVERY ms, up)
| restart (kill the process, start a new one on the same saves) | # comment"""
import subprocess, sys, shutil, glob, os
from PIL import Image
ROOT = "/home/user/porthole/apps/porthole"
FIX = "/home/user/porthole/.superpowers/sdd/2026-11-08-pets-club-shelter/fixture"
trace, out = sys.argv[1], sys.argv[2]
os.makedirs(out, exist_ok=True)
if "--keep" not in sys.argv:
    for f in glob.glob(ROOT + "/build/host/*.sav"): os.remove(f)
    for f in glob.glob(FIX + "/*.sav"): shutil.copy(f, ROOT + "/build/host/")
def start(): return subprocess.Popen([ROOT + "/build/host/snap", "--serve"], cwd=ROOT, stdin=subprocess.PIPE, stdout=subprocess.PIPE)
p = start()
def cmd(c):
    p.stdin.write((c + "\n").encode()); p.stdin.flush()
    l = p.stdout.readline()
    if c == "frame":
        n = int(l.split()[1]); data = p.stdout.read(n)
        return data
    return l
def frame(name):
    d = cmd("frame")
    # parse P6
    img = Image.frombytes("RGB", (480, 480), d)
    img.save(f"{out}/{name}.png"); print("frame", name)
t = 0
for raw in open(trace):
    line = raw.split("#")[0].strip()
    if not line: continue
    a = line.split()
    if a[0] == "frame": frame(a[1])
    elif a[0] == "kill":
        p.kill(); p.wait(); p = start(); print("kill")
    elif a[0] == "restart":
        p.stdin.close(); p.wait(); p = start(); print("restart")
    elif a[0] == "press":
        x, y, ms = a[1], a[2], int(a[3]); every = int(a[4]) if len(a) > 4 else 0
        tag = a[5] if len(a) > 5 else f"press{x}_{y}_{ms}"
        cmd(f"down {x} {y}"); nf = max(1, round(ms / 33)); ev = round(every / 33) if every else 0
        for i in range(1, nf):
            cmd("tick 33")
            if ev and i % ev == 0: frame(f"{tag}_{(i+1)*33:04d}ms")
        cmd("up")
    else:
        cmd(line)
p.stdin.close(); p.wait()
