#!/bin/bash
# usage: send.sh "cmd1" "cmd2" ... ; shortcuts: "tap X Y" -> down,tick 100,up ; "press X Y MS"
S=/tmp/claude-0/-home-user-porthole/4e7b951a-0e4f-5eae-859d-2ab6ed697cd9/scratchpad/sim.sock
lines=""
for c in "$@"; do
  set -- $c
  case $1 in
    tap) lines+="down $2 $3"$'\n'"tick 100"$'\n'"up"$'\n'"tick 33"$'\n';;
    press) lines+="down $2 $3"$'\n'"tick $4"$'\n'"up"$'\n'"tick 33"$'\n';;
    *) lines+="$c"$'\n';;
  esac
done
printf "%s" "$lines" | python3 -I -c "
import socket,sys
s=socket.socket(socket.AF_UNIX);s.connect('$S');f=s.makefile('rwb')
for l in sys.stdin:
    f.write(l.encode());f.flush();print(l.strip(),'->',f.readline().decode().strip())
"
